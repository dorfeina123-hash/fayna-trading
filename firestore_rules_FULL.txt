rules_version = '2';
// ═══════════════════════════════════════════════════════════════
// Fayna Trading — חוקי Firestore מלאים (v24)
// Firebase Console → Firestore Database → Rules → הדבק → Publish
//
// עיקרון מנחה: כל משתמש ניגש אך ורק למסמך שלו.
// שינוי ה-URL או קריאה ישירה ל-API לא יאפשרו גישה למידע של אחר.
// ═══════════════════════════════════════════════════════════════
service cloud.firestore {
  match /databases/{database}/documents {

    // ── עזרים ──
    function signedIn()  { return request.auth != null; }
    function isSelf(uid) { return signedIn() && request.auth.uid == uid; }
    function isAdmin()   {
      return signedIn() &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true;
    }
    // שדות שמשתמש רגיל לא רשאי לשנות בעצמו (הסלמת הרשאות / עקיפת תשלום)
    function noPrivilegeEscalation() {
      return !request.resource.data.diff(resource.data).affectedKeys()
              .hasAny(['isAdmin','plan','planExpiresAt','planStatus','planCycle',
                       'subscriptionPlan','subscriptionStatus','paymentStatus',
                       'featureOverrides','status','refRewardsGranted']);
    }

    // ── משתמשים: כל אחד רואה ועורך רק את עצמו ──
    match /users/{uid} {
      allow read:   if isSelf(uid) || isAdmin();
      allow create: if isSelf(uid);
      allow update: if isAdmin() || (isSelf(uid) && noPrivilegeEscalation());
      allow delete: if isSelf(uid) || isAdmin();

      // תת-אוספים של נתוני מסחר — בבעלות המשתמש בלבד
      match /{document=**} {
        allow read, write: if isSelf(uid) || isAdmin();
      }
    }

    // ── הפניות ──
    match /referrals/{refereeUid} {
      allow create, update: if isSelf(refereeUid);
      allow read: if signedIn() &&
                  (resource.data.referrerUid == request.auth.uid ||
                   request.auth.uid == refereeUid || isAdmin());
      allow delete: if isSelf(refereeUid) || isAdmin();
    }

    // מיפוי קוד הזמנה קצר → מזהה משתמש (הקישור לא חושף את המזהה)
    match /refCodes/{code} {
      allow read: if true;                       // נדרש בזמן הרשמה, לפני התחברות
      allow create, update: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // מונה קליקים על קישור הזמנה (אנונימי)
    match /refClicks/{code} {
      allow read: if signedIn();
      allow create, update: if request.resource.data.keys().hasOnly(['count','last']);
      allow delete: if isAdmin();
    }

    // ── דיוור והסרות ──
    match /dnc/{emailKey} {
      allow read:   if signedIn();               // בדיקה לפני שליחה ובהרשמה חוזרת
      allow create: if signedIn();               // הסרה עצמית
      allow update, delete: if isAdmin();
    }
    match /mailingLogs/{logId} {
      allow create: if signedIn();
      allow read:   if isAdmin();
      allow update, delete: if false;            // יומן — לא ניתן לעריכה
    }

    // ── חיוב ותשלומים ──
    match /billing_requests/{reqId} {
      allow create: if true;                     // גם מבקר לא רשום מעמוד התמחור
      allow read:   if isAdmin();
      allow update, delete: if isAdmin();
    }

    // ── הגדרות מערכת ──
    match /settings/{doc} {
      allow read:  if signedIn();
      allow write: if isAdmin();
    }

    // ── לוגים ותפעול ──
    match /loginLogs/{logId} {
      allow create: if signedIn();
      allow read:   if isAdmin();
      allow update, delete: if false;
    }
    match /adminLogs/{logId} {
      allow create: if isAdmin();
      allow read:   if isAdmin();
      allow update, delete: if false;
    }
    match /waitlist/{docId} {
      allow create: if signedIn();
      allow read, update, delete: if isAdmin();
    }
    match /invites/{email} {
      allow read:   if signedIn();
      allow write:  if isAdmin();
    }

    // ── עורך ויזואלי (Visual Builder) ──
    // העיצוב שפורסם נקרא על ידי כל משתמש; כתיבה — אדמין בלבד.
    match /site_design/{doc} {
      allow read:  if true;
      allow write: if isAdmin();
    }
    match /site_design_versions/{verId} {
      allow read:   if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;   // היסטוריה — לא ניתנת לשינוי
    }

    // ── ברירת מחדל: סגור ──
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
